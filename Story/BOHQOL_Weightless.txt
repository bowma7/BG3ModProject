// ============================================================
// WEIGHTLESS HANDLING
//
// Items inside the BOHQOL Bag hierarchy receive
// BOHQOL_TestWeightless.
//
// Weight restoration is scoped through
// DB_BOHQOL_WeightlessExitPending.
//
// An item enters that DB only after it is removed from a known
// BOHQOL container. This avoids querying HasPassive against
// every unrelated AddedTo event in the world.
// ============================================================


// ============================================================
// BAG ROOT
// ============================================================

IF
AddedTo(_Item, _Bag, _AddType)
AND
DB_BOHQOL_BagTarget((GUIDSTRING)_Player, (GUIDSTRING)_Bag)
AND
IsItem(_Item, 1)
AND
HasPassive(_Item, "BOHQOL_TestWeightless", 0)
THEN
AddPassive(_Item, "BOHQOL_TestWeightless");


IF
AddedTo(_Item, _Bag, _AddType)
AND
DB_BOHQOL_BagTarget((GUIDSTRING)_Player, (GUIDSTRING)_Bag)
AND
DB_BOHQOL_WeightlessExitPending(_Item)
THEN
NOT DB_BOHQOL_WeightlessExitPending(_Item);

// ============================================================
// BOHQOL EXIT TRACKING
//
// RemovedFrom is emitted when an object is removed from an
// inventory. Only removals from cached BOHQOL containers create
// an exit-pending fact.
//
// Moving between BOHQOL containers temporarily creates the fact;
// the destination AddedTo rules above immediately clear it.
//
// An actual move to loose character inventory is handled by the
// final resolver below.
// ============================================================

IF

AND
DB_BOHQOL_BagTarget((GUIDSTRING)_Player, (GUIDSTRING)_Bag)
THEN

// ============================================================
// EXIT -> LOOSE CHARACTER INVENTORY
//
// Unlike the previous implementation, this AddedTo rule joins
// against a tiny BOHQOL-owned DB before performing any passive
// query.
//
// Unrelated world AddedTo events therefore have no
// DB_BOHQOL_WeightlessExitPending fact and stop here.
//
// GetDirectInventoryOwner reconfirms that the item really ended
// up loose in the character's inventory.
// ============================================================

IF
AddedTo(_Item, _Player, _AddType)
AND
DB_BOHQOL_WeightlessExitPending(_Item)
AND
IsCharacter(_Player, 1)
AND
GetDirectInventoryOwner(_Item, _Player)
AND
HasPassive(_Item, "BOHQOL_TestWeightless", 1)
THEN
RemovePassive(_Item, "BOHQOL_TestWeightless");
NOT DB_BOHQOL_WeightlessExitPending(_Item);
