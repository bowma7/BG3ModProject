// ============================================================
// BOHQOL_BagTargets - REWRITE
//
// Architecture:
//   CACHE VALIDATION -> ENSURE -> DISCOVER/RECOVER
//
// Principles:
//   - Target DBs are caches.
//   - Valid cached targets are reused.
//   - Discovery runs only when a target is missing/stale.
//   - Exact hierarchy is validated with direct ownership.
//   - Sorting policy does NOT live in this file.
//   - Weightless repair is kept separate from target discovery.
//   - Retrieval reuses the same target cache.
// ============================================================

// ============================================================
// VALID TARGET QUERIES
//
// These queries validate the exact cached hierarchy.
//
// Root Bag:
//   Bag -> Player
//
// Equipment branch:
//   Equipment -> Bag
//   Weapon / Armor / Accessories -> Equipment
//
// Consumables branch:
//   Consumables -> Bag
//   Potion / Scroll / Arrow / Throwable -> Consumables
//
// Direct Bag children:
//   Alchemy / Books / Key Ring / Quest / Misc -> Bag
//
// The parent target is included in each nested query, so moving a
// BOHQOL organizer elsewhere in the player's inventory invalidates
// the cache rather than silently continuing to use the wrong tree.
// ============================================================

PROC_BOHQOL_EnsureBagTarget((GUIDSTRING)_Player)
AND
QRY_BOHQOL_HasValidBagTarget(_Player)
THEN
DB_NOOP(1);


PROC

AND
NOT QRY_BOHQOL_HasValidBagTarget(_Player)

// ============================================================
// BAG OPEN TARGET REFRESH
//
// Opening a specific BOHQOL Bag makes that runtime Bag the active
// target for the player. This preserves the useful behavior of the
// old TemplateOpening path, especially if more than one replacement
// Bag exists.
//
// Child caches do not need to be globally cleared here. Their own
// validity queries include the active parent, so they invalidate
// naturally if the active Bag changes.
// ============================================================

IF
TemplateOpening(
    (ITEMROOT)BOHQOL_BagOfHolding_568c5b36-5f4f-4f47-bcb9-2c8fdbf65d90,
    _Bag,
    _Player
)
AND
GetDirectInventoryOwner(_Bag, _Player)
THEN
PROC_BOHQOL_RefreshOpenedBagTarget(
    (GUIDSTRING)_Player,
    (GUIDSTRING)_Bag
);

PROC_BOHQOL_EnsureAllSortTargets(
    (GUIDSTRING)_Player
);

PROC_BOHQOL_RepairCachedContainerWeightless(
    (GUIDSTRING)_Player
);


PROC
PROC_BOHQOL_RefreshOpenedBagTarget(
    (GUIDSTRING)_Player,
    (GUIDSTRING)_Bag
)
THEN
PROC_BOHQOL_ClearBagTarget(_Player);
DB_BOHQOL_BagTarget((GUIDSTRING)_Player, (GUIDSTRING)_Bag);

// ============================================================
// GENERIC RETRIEVAL
//
// Retrieval is intentionally separate from sorting.
//
// The target's top-level inventory owner determines the player,
// so category-specific temporary retrieval DBs are unnecessary.
//
// Weightlessness is removed BEFORE moving. This is important for
// Keys and Alchemy because BG3 may immediately route those items
// into a vanilla Keychain / Alchemy Pouch instead of leaving them
// loose at the player root.
// ============================================================

PROC
PROC_BOHQOL_RetrieveTarget((GUIDSTRING)_Target)
THEN
IterateInventory(
    _Target,
    "BOHQOL_RetrieveTarget_Item",
    ""
);


IF
EntityEvent(_Item, "BOHQOL_RetrieveTarget_Item")
AND
IsItem(_Item, 1)
AND
GetDirectInventoryOwner(_Item, _Target)
AND
GetInventoryOwner(_Target, _Player)
AND
IsCharacter(_Player, 1)
AND
GetStackAmount((ITEM)_Item, _Amount, _)
THEN
RemovePassive(
    _Item,
    "BOHQOL_TestWeightless"
);

ToInventory(
    _Item,
    _Player,
    _Amount,
    0,
    0
);

// Existing vanilla Keychain is already directly on player.
PROC
PROC_BOHQOL_PrepareRetrieveKeysStage((GUIDSTRING)_Player)
AND
GetItemByTemplateInInventory(
    (ITEMROOT)ee329627-dbee-405f-b9a6-b260de9ad34c,
    _Player,
    _VanillaKeychain
)
AND
GetDirectInventoryOwner(_VanillaKeychain, _Player)
THEN
PROC_BOHQOL_RetrieveKeys(_Player);


// Existing vanilla Keychain is nested.
// Move the Keychain itself to player root, then continue when
// TemplateAddedTo confirms the move.
PROC
PROC_BOHQOL_PrepareRetrieveKeysStage((GUIDSTRING)_Player)
AND
GetItemByTemplateInInventory(
    (ITEMROOT)ee329627-dbee-405f-b9a6-b260de9ad34c,
    _Player,
    _VanillaKeychain
)
AND
NOT GetDirectInventoryOwner(_VanillaKeychain, _Player)
THEN
DB_BOHQOL_RetrieveKeySafetyWaiting(_Player);

ToInventory(
    _VanillaKeychain,
    _Player,
    1,
    0,
    0
);


// No vanilla Keychain exists anywhere in the player's inventory.
PROC
PROC_BOHQOL_PrepareRetrieveKeysStage((GUIDSTRING)_Player)
AND
NOT GetItemByTemplateInInventory(
    (ITEMROOT)ee329627-dbee-405f-b9a6-b260de9ad34c,
    _Player,
    _
)
THEN
DB_BOHQOL_RetrieveKeySafetyWaiting(_Player);

TemplateAddTo(
    (ITEMROOT)ee329627-dbee-405f-b9a6-b260de9ad34c,
    _Player,
    1,
    0
);


IF
TemplateAddedTo(
    (ROOT)ee329627-dbee-405f-b9a6-b260de9ad34c,
    _VanillaKeychain,
    _Player,
    _AddType
)
AND
DB_BOHQOL_RetrieveKeySafetyWaiting(_Player)
AND
GetDirectInventoryOwner(_VanillaKeychain, _Player)
THEN
NOT DB_BOHQOL_RetrieveKeySafetyWaiting(_Player);

PROC_BOHQOL_RetrieveKeys(_Player);
