// ============================================================
// BOHQOL_SortWeapons - REWRITE
//
// This goal owns sorting CLASSIFICATION / POLICY / MOVEMENT.
//
// Architecture:
//   ACTION
//     -> target ensurance (BagTargets)
//     -> source/context
//     -> ordered policy dispatcher
//     -> shared category classifier
//     -> shared movement
//
// Manual, FULL, SAFE and Item Enter do NOT maintain separate
// copies of category classification.
//
// FULL and SAFE differ only in policy:
//   SAFE protects Throwables and Quest & Important.
//
// Item Enter differs only in source:
//   broad/manual source = loose player inventory
//   Item Enter source   = BOHQOL root Bag
// ============================================================


// ============================================================
// PRECEDENCE / CLASSIFICATION HELPERS
//
// FULL priority:
//   1. Key
//   2. Quest / Important
//   3. Spell Scroll
//   4. Book / Note
//   5. Weapon
//   6. Armor
//   7. Accessory
//   8. Potion / Coating
//   9. Arrow
//  10. Throwable
//  11. Alchemy
//  12. Misc
//
// SAFE:
//   Key remains allowed.
//   Quest / Important and Throwable are protected and block
//   lower-priority fallthrough.
// ============================================================

// ============================================================
// TRANSIENT PRECEDENCE CLAIM
//
// Osiris PROC calls are ordered, but movement side effects should
// not be used as the only mechanism preventing a later category
// from processing the same item.
//
// The dispatcher clears this transient fact before processing.
// The first matching category claims the item before moving it.
// Every lower-priority category then fails immediately.
//
// The fact is cleared again after dispatch so it never becomes
// persistent classification state.
// ============================================================

PROC
PROC_BOHQOL_ClearSortClaim((GUIDSTRING)_Item)
AND
DB_BOHQOL_SortClaimed(_Item)
THEN
NOT DB_BOHQOL_SortClaimed(_Item);


QRY
QRY_BOHQOL_SortItemUnclaimed((GUIDSTRING)_Item)
AND
NOT DB_BOHQOL_SortClaimed(_Item)
THEN
DB_NOOP(1);


// ============================================================
// SHARED MOVEMENT
//
// Weightless is intentionally NOT applied here.
// BOHQOL_Weightless owns the AddedTo behavior for managed targets.
// ============================================================

PROC
PROC_BOHQOL_MoveItemToTarget(
    (GUIDSTRING)_Item,
    (GUIDSTRING)_Target
)
AND
GetStackAmount((ITEM)_Item, _Amount, _)
THEN
ToInventory(
    _Item,
    _Target,
    _Amount,
    0,
    0
);

// ============================================================
// FULL DISPATCHER
//
// Calls are deliberately ordered. Once a higher-priority category
// moves the item, later category processors fail their direct-owner
// source guard.
// ============================================================

PROC
PROC_BOHQOL_ProcessFullItem(
    (GUIDSTRING)_Item,
    (GUIDSTRING)_Player,
    (GUIDSTRING)_Source
)
THEN
PROC_BOHQOL_ClearSortClaim(_Item);
PROC_BOHQOL_ProcessKeyItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessQuestItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessScrollItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessBookItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessWeaponItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessArmorItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessAccessoryItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessPotionItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessArrowItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessThrowableItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessAlchemyItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessMiscItem(_Item, _Player, _Source);
PROC_BOHQOL_ClearSortClaim(_Item);


// ============================================================
// SAFE DISPATCHER
//
// Quest/Important and Throwable are protected before any SAFE
// category can run. Key remains allowed because QuestImportant
// explicitly excludes Keys and Key has highest precedence.
// ============================================================

PROC
PROC_BOHQOL_ProcessSafeItem(
    (GUIDSTRING)_Item,
    (GUIDSTRING)_Player,
    (GUIDSTRING)_Source
)
AND
NOT QRY_BOHQOL_IsSafeProtected(_Item)
THEN
PROC_BOHQOL_ClearSortClaim(_Item);
PROC_BOHQOL_ProcessKeyItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessScrollItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessBookItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessWeaponItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessArmorItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessAccessoryItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessPotionItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessArrowItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessAlchemyItem(_Item, _Player, _Source);
PROC_BOHQOL_ProcessMiscItem(_Item, _Player, _Source);
PROC_BOHQOL_ClearSortClaim(_Item);


// ============================================================
// FULL SORT - BROAD PLAYER INVENTORY
// ============================================================

PROC
PROC_BOHQOL_SortAll((GUIDSTRING)_Player)
THEN
PROC_BOHQOL_EnsureAllSortTargets(_Player);

IterateInventory(
    _Player,
    "BOHQOL_SortFull_Item",
    ""
);


IF
EntityEvent(_Item, "BOHQOL_SortFull_Item")
AND
GetDirectInventoryOwner(_Item, _Player)
AND
IsCharacter(_Player, 1)
THEN
PROC_BOHQOL_ProcessFullItem(
    (GUIDSTRING)_Item,
    (GUIDSTRING)_Player,
    (GUIDSTRING)_Player
);

// ============================================================
// MANUAL CATEGORY ACTIONS
//
// Manual actions use the SAME category processors as FULL/SAFE.
// The only differences are:
//   - ensure only the required target;
//   - iterate with only the requested category event.
//
// Automatic sort mode never blocks these procedures.
// ============================================================


// ------------------------------------------------------------
// WEAPONS
// ------------------------------------------------------------

PROC
PROC_BOHQOL_SortWeapons((GUIDSTRING)_Player)
THEN
PROC_BOHQOL_EnsureWeaponTarget(_Player);

IterateInventory(
    _Player,
    "BOHQOL_SortWeapon_Item",
    ""
);


IF
EntityEvent(_Item, "BOHQOL_SortWeapon_Item")
AND
GetDirectInventoryOwner(_Item, _Player)
AND
IsCharacter(_Player, 1)
THEN
PROC_BOHQOL_ClearSortClaim(_Item);
PROC_BOHQOL_ProcessWeaponItem(_Item, _Player, _Player);
PROC_BOHQOL_ClearSortClaim(_Item);
