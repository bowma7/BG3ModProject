// ============================================================
// BAG ACTIONS
//
// This file determines WHAT manual action the player requested.
//
// Target ensurance/cache belongs to BOHQOL_BagTargets.
// Classification/policy/movement belongs to BOHQOL_SortWeapons.
//
// Manual actions call the final public action entry points directly.
// They are intentionally independent of FULL / SAFE / DISABLED
// automatic sort mode.
// ============================================================


// ============================================================
// BAG ACTION: ALL - SORT MODE
// ============================================================

IF
UsingSpell(
    _Caster,
    "Shout_BOHQOL_BagAction_All",
    _SpellType,
    _SpellElement,
    _StoryActionID
)
AND
HasActiveStatus(
    _Caster,
    "BOHQOL_ACTIONMODE_SORT",
    1
)
THEN
PROC_BOHQOL_SortAll((GUIDSTRING)_Caster);
